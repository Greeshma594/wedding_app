<script>
let currentYear  = new Date().getFullYear();
let currentMonth = new Date().getMonth();
let selectedDate = null;
let bookedDates  = {};

const pad = n => String(n).padStart(2, '0');
const toISO = (y, m, d) => `${y}-${pad(m+1)}-${pad(d)}`;

function formatDateFull(iso) {
  const [y, m, d] = iso.split('-').map(Number);
  return new Date(y, m - 1, d).toLocaleDateString('en-US', {
    weekday: 'long', year: 'numeric', month: 'long', day: 'numeric'
  });
}
function formatDateShort(iso) {
  const [y, m, d] = iso.split('-').map(Number);
  return new Date(y, m - 1, d).toLocaleDateString('en-US', {
    month: 'short', day: 'numeric', year: 'numeric'
  });
}

async function fetchBookedDates() {
  const res = await fetch('/api/bookings/booked-dates');
  bookedDates = await res.json();
}
async function fetchBookingsForDate(dateISO) {
  const res = await fetch(`/api/bookings/date/${dateISO}`);
  return res.json();
}
async function deleteBooking(id) {
  const res = await fetch(`/api/bookings/${id}`, { method: 'DELETE' });
  return res.json();
}

function showToast(msg, type = 'success') {
  const container = document.getElementById('toastContainer');
  const toast = document.createElement('div');
  toast.className = `toast ${type}`;
  toast.innerHTML = `<span class="toast-icon">${type === 'success' ? '✓' : '✕'}</span> ${msg}`;
  container.appendChild(toast);
  requestAnimationFrame(() => { requestAnimationFrame(() => { toast.classList.add('show'); }); });
  setTimeout(() => {
    toast.classList.remove('show');
    setTimeout(() => toast.remove(), 350);
  }, 3200);
}

function renderCalendar() {
  const grid  = document.getElementById('calGrid');
  const label = document.getElementById('monthLabel');

  label.textContent = new Date(currentYear, currentMonth, 1)
    .toLocaleDateString('en-US', { month: 'long', year: 'numeric' });

  const firstDow  = new Date(currentYear, currentMonth, 1).getDay();
  const daysCount = new Date(currentYear, currentMonth + 1, 0).getDate();
  const prevCount = new Date(currentYear, currentMonth,     0).getDate();

  const today    = new Date();
  const todayISO = toISO(today.getFullYear(), today.getMonth(), today.getDate());

  grid.innerHTML = '';

  for (let i = firstDow - 1; i >= 0; i--) {
    const cell = document.createElement('div');
    cell.className = 'cal-day other-month';
    cell.innerHTML = `<span class="day-num">${prevCount - i}</span>`;
    grid.appendChild(cell);
  }

  for (let d = 1; d <= daysCount; d++) {
    const iso  = toISO(currentYear, currentMonth, d);
    const cell = document.createElement('div');
    cell.className = 'cal-day';
    cell.dataset.date = iso;

    const count = bookedDates[iso] || 0;
    if (count)          cell.classList.add('has-bookings');
    if (iso === todayISO)    cell.classList.add('today');
    if (iso === selectedDate) cell.classList.add('selected');

    const dotsHtml = count
      ? `<div class="booking-dots">${
          Array.from({ length: Math.min(count, 3) }, () => '<span class="dot"></span>').join('')
          + (count > 3 ? `<span class="dot-more">+${count-3}</span>` : '')
        }</div>`
      : '';

    cell.innerHTML = `<span class="day-num">${d}</span>${dotsHtml}`;
    cell.addEventListener('click', () => selectDate(iso));
    grid.appendChild(cell);
  }

  const total    = firstDow + daysCount;
  const trailing = total % 7 === 0 ? 0 : 7 - (total % 7);
  for (let d = 1; d <= trailing; d++) {
    const cell = document.createElement('div');
    cell.className = 'cal-day other-month';
    cell.innerHTML = `<span class="day-num">${d}</span>`;
    grid.appendChild(cell);
  }
}

async function selectDate(iso) {
  selectedDate = iso;
  renderCalendar();

  document.getElementById('noDateState').style.display  = 'none';
  document.getElementById('sideHeader').style.display   = 'block';
  document.getElementById('sideContent').style.display  = 'block';
  document.getElementById('addBtnWrap').style.display   = 'block';
  document.getElementById('sideDateLabel').textContent  = formatDateFull(iso);

  const content = document.getElementById('sideContent');
  content.innerHTML = '<div class="loading-wrap"><div class="spinner"></div></div>';

  const bookings = await fetchBookingsForDate(iso);
  document.getElementById('sideSubtitle').textContent = bookings.length === 0
    ? 'No bookings yet'
    : `${bookings.length} booking${bookings.length > 1 ? 's' : ''}`;

  renderBookingsList(bookings, content);
}

function renderBookingsList(bookings, container) {
  if (bookings.length === 0) {
    container.innerHTML = `
      <div class="empty-state">
        <div class="empty-icon">💍</div>
        <p>No bookings for this date yet.<br>Click <strong>Add New Booking</strong> below to get started.</p>
      </div>`;
    return;
  }

  container.innerHTML = '';
  bookings.forEach(b => {
    const card = document.createElement('div');
    card.className = 'booking-card';

    const imgSection = b.imageUrl
      ? `<img class="booking-card-img" src="${b.imageUrl}" alt="${b.clientName}" />`
      : `<div class="booking-card-img-placeholder">💐</div>`;

    const sameDayStr = b.startDate === b.endDate
      ? formatDateShort(b.startDate)
      : `${formatDateShort(b.startDate)} → ${formatDateShort(b.endDate)}`;

    card.innerHTML = `
      ${imgSection}
      <div class="booking-card-body">
        <div class="booking-card-title">${escHtml(b.clientName)}</div>
        <span class="booking-card-type">${escHtml(b.eventType)}</span>
        <div class="booking-meta">
          <span><span class="meta-icon">📅</span>${sameDayStr}</span>
          ${b.contactEmail ? `<span><span class="meta-icon">✉️</span>${escHtml(b.contactEmail)}</span>` : ''}
          ${b.contactPhone ? `<span><span class="meta-icon">📞</span>${escHtml(b.contactPhone)}</span>` : ''}
          ${b.notes ? `<span><span class="meta-icon">📝</span>${escHtml(b.notes.substring(0,80))}${b.notes.length>80?'…':''}</span>` : ''}
        </div>
      </div>
      <div class="booking-card-actions">
        <button class="btn-sm view"   onclick="openEditModal('${b.id}')">✏️ Edit</button>
        <button class="btn-sm delete" onclick="confirmDelete('${b.id}', '${escHtml(b.clientName)}')">🗑️ Delete</button>
      </div>
    `;
    container.appendChild(card);
  });
}

function escHtml(str) {
  return String(str)
    .replace(/&/g,'&amp;').replace(/</g,'&lt;')
    .replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

async function confirmDelete(id, name) {
  if (!confirm(`Delete the booking for "${name}"? This cannot be undone.`)) return;
  try {
    await deleteBooking(id);
    showToast('Booking deleted.', 'success');
    await fetchBookedDates();
    renderCalendar();
    if (selectedDate) selectDate(selectedDate);
  } catch {
    showToast('Could not delete booking.', 'error');
  }
}

const overlay   = document.getElementById('modalOverlay');
const form      = document.getElementById('bookingForm');
const editIdFld = document.getElementById('editBookingId');

function openAddModal() {
  editIdFld.value = '';
  form.reset();
  clearImagePreview();
  document.getElementById('existingImageWrap').style.display = 'none';
  if (selectedDate) {
    document.getElementById('fStartDate').value = selectedDate;
    document.getElementById('fEndDate').value   = selectedDate;
  }
  document.getElementById('modalTitle').textContent    = 'New Booking';
  document.getElementById('modalSubtitle').textContent = selectedDate ? `For ${formatDateFull(selectedDate)}` : '';
  document.getElementById('saveBtn').textContent       = '💐 Save Booking';
  overlay.classList.add('visible');
  document.getElementById('fClientName').focus();
}

async function openEditModal(id) {
  try {
    const res = await fetch(`/api/bookings/${id}`);
    const b   = await res.json();
    editIdFld.value = id;
    document.getElementById('fClientName').value = b.clientName;
    document.getElementById('fEventType').value  = b.eventType;
    document.getElementById('fStartDate').value  = b.startDate;
    document.getElementById('fEndDate').value    = b.endDate;
    document.getElementById('fEmail').value      = b.contactEmail || '';
    document.getElementById('fPhone').value      = b.contactPhone || '';
    document.getElementById('fNotes').value      = b.notes || '';
    clearImagePreview();
    if (b.imageUrl) {
      document.getElementById('existingImg').src             = b.imageUrl;
      document.getElementById('existingImageWrap').style.display = 'block';
    } else {
      document.getElementById('existingImageWrap').style.display = 'none';
    }
    document.getElementById('modalTitle').textContent    = 'Edit Booking';
    document.getElementById('modalSubtitle').textContent = `${b.clientName} · ${b.eventType}`;
    document.getElementById('saveBtn').textContent       = '💾 Update Booking';
    overlay.classList.add('visible');
  } catch {
    showToast('Could not load booking details.', 'error');
  }
}

function closeModal() { overlay.classList.remove('visible'); }

function clearImagePreview() {
  document.getElementById('uploadPreview').style.display = 'none';
  document.getElementById('previewImg').src = '';
  document.getElementById('fImage').value  = '';
  document.getElementById('uploadZone').style.display = 'block';
}

document.getElementById('fImage').addEventListener('change', function () {
  const file = this.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = e => {
    document.getElementById('previewImg').src              = e.target.result;
    document.getElementById('uploadPreview').style.display = 'inline-block';
    document.getElementById('uploadZone').style.display    = 'none';
  };
  reader.readAsDataURL(file);
});

document.getElementById('removeImageBtn').addEventListener('click', clearImagePreview);

const zone = document.getElementById('uploadZone');
zone.addEventListener('dragover',  e => { e.preventDefault(); zone.classList.add('drag-over'); });
zone.addEventListener('dragleave', () => zone.classList.remove('drag-over'));
zone.addEventListener('drop', e => {
  e.preventDefault();
  zone.classList.remove('drag-over');
  if (e.dataTransfer.files.length) {
    document.getElementById('fImage').files = e.dataTransfer.files;
    document.getElementById('fImage').dispatchEvent(new Event('change'));
  }
});

form.addEventListener('submit', async function (e) {
  e.preventDefault();
  const saveBtn = document.getElementById('saveBtn');
  saveBtn.disabled    = true;
  saveBtn.textContent = 'Saving…';

  const fd = new FormData();
  fd.append('clientName',   document.getElementById('fClientName').value.trim());
  fd.append('eventType',    document.getElementById('fEventType').value);
  fd.append('startDate',    document.getElementById('fStartDate').value);
  fd.append('endDate',      document.getElementById('fEndDate').value);
  fd.append('contactEmail', document.getElementById('fEmail').value.trim());
  fd.append('contactPhone', document.getElementById('fPhone').value.trim());
  fd.append('notes',        document.getElementById('fNotes').value.trim());

  const imageFile = document.getElementById('fImage').files[0];
  if (imageFile) fd.append('image', imageFile);

  const id     = editIdFld.value;
  const url    = id ? `/api/bookings/${id}` : '/api/bookings';
  const method = id ? 'PUT' : 'POST';

  try {
    const res  = await fetch(url, { method, body: fd });
    const data = await res.json();
    if (!res.ok) throw new Error(data.error || 'Server error');
    showToast(id ? 'Booking updated!' : 'Booking saved!', 'success');
    closeModal();
    await fetchBookedDates();
    renderCalendar();
    if (selectedDate) selectDate(selectedDate);
  } catch (err) {
    showToast(err.message || 'Could not save booking.', 'error');
  } finally {
    saveBtn.disabled    = false;
    saveBtn.textContent = id ? '💾 Update Booking' : '💐 Save Booking';
  }
});

  currentMonth--;

  renderCalendar();
});
document.getElementById('nextBtn').addEventListener('click', () => {
  currentMonth++;
  if (currentMonth > 11) { currentMonth = 0; currentYear++; }
  renderCalendar();
});
document.getElementById('todayBtn').addEventListener('click', () => {
  const today  = new Date();
  currentYear  = today.getFullYear();
  currentMonth = today.getMonth();
  selectedDate = toISO(currentYear, currentMonth, today.getDate());
  renderCalendar();
  selectDate(selectedDate);
});
